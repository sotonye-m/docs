  Bearer token: `Bearer YOUR_ACCESS_TOKEN`
</ParamField>

<ParamField path="customer_id" type="string" required>

</ParamField>

<ParamField body="username" type="string">Display name.</ParamField>

<ParamField body="email" type="string">Email address.</ParamField>

<ParamField body="phone_number" type="string">Phone number.</ParamField>

<RequestExample>
```bash
curl -X PATCH 'https://crypto-api.yativo.com/api/v1/customers/6650a1f2c3d4e5f6a7b8c9d0' \
  -H 'Authorization: Bearer YOUR_ACCESS_TOKEN' \
  -H 'Content-Type: application/json' \
  -d '{ "phone_number": "+14155550100" }'
```
</RequestExample>
